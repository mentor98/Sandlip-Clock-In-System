        │
        └─→ Cron job triggers
            │
            └─→ absenceMarker.js
                │
                ├─ markAbsent()
                │   ├─ Get today's date
                │   ├─ Query sessions started today
                │   │   └─ attendance_sessions WHERE DATE(started_at) = TODAY

API ENDPOINT FLOW:
═══════════════════════════════════════════════════════════════════════════

GET /api/admin/attendance/absent?session_id=<uuid>&from=<date>&to=<date>
│
├─ Middleware:
│  ├─ requireAuth() ✓
│  └─ requireAdmin() ✓
│
├─ Query database:
│   └─ SELECT * FROM attendance
│       WHERE verification_status = 'AUTO_ABSENT'
│       AND (session_id = <uuid> if provided)
│       AND (marked_absent_at >= <from> if provided)
│       AND (marked_absent_at <= <to> if provided)
│
├─ JOIN with related data:
│   ├─ JOIN students ON attendance.student_id = students.id
│   └─ JOIN locations ON attendance.location_id = locations.id
│
└─ Return JSON:
    {
      "absent": [
        {
          "id": "uuid",
          "student_id": "uuid",
          "students": { "full_name": "John Doe", "student_id": "STU-001" },
          "locations": { "name": "Main Hall" },
          "verification_status": "AUTO_ABSENT",
          "absence_reason": "No clock-in by 09:30 AM",
          "marked_absent_at": "2024-09-01T09:30:00Z"
        }
      ]
    }


DATABASE SCHEMA (Relevant Tables):
═══════════════════════════════════════════════════════════════════════════

attendance_sessions
├─ id: uuid
├─ title: text
├─ status: text (ACTIVE | CLOSED | EXPIRED)
├─ on_time_until: time default 09:00 ← NEW
├─ started_at: timestamptz
└─ created_at: timestamptz

attendance
├─ id: uuid
├─ student_id: uuid [FK → students]
├─ session_id: uuid [FK → attendance_sessions]
├─ type: text (clock_in | clock_out | absence)
├─ recorded_at: timestamp
├─ verification_status: text (VERIFIED | REVIEW | REJECTED | AUTO_ABSENT) ← UPDATED
├─ is_late: boolean ← NEW
├─ marked_absent_at: timestamptz ← NEW
└─ absence_reason: text ← NEW

audit_log
├─ id: uuid
├─ student_id: uuid [FK → students]
├─ event_type: text (student_marked_absent | ...) ← NEW EVENT TYPE
└─ detail: jsonb
    └─ {
        "session_id": "uuid",
        "session_title": "Morning Session",
        "reason": "No clock-in by 09:30 AM",
        "marked_by": "scheduler"
      }
